# a moderator adds and edits a tag, then edits and bans a domain (the domain pages are 404 on both trees: their `constraints id: DOMAINS_IDENTIFIER` regexp is outside the router subset, a reported error)
#! login alice@example.com
GET /mod/tags/new
POST /mod/tags tag[tag]=replay&tag[category_name]=general&tag[description]=Made+by+the+replay&tag[hotness_mod]=0
GET /t/replay
GET /mod/tags/replay/edit
POST /mod/tags/replay _method=patch&tag[description]=Edited+by+the+replay&tag[category_name]=general
GET /tags
GET /domains/example.com -> 404
GET /mod/domains/example.com/edit -> 404
POST /mod/domains_ban/example.com _method=patch&domain[banned_reason]=Replay+ban -> 404
GET /domains/example.com -> 404
